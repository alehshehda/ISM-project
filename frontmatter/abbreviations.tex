\UnnumberedChapter{\abbrevname}

\begin{abbreviations}
    \item[AML]     Anti-Money Laundering (przeciwdziałanie praniu pieniędzy)
    \item[API]     Application Programming Interface (interfejs programistyczny aplikacji)
    \item[ATO]     Account Takeover (przejęcie konta użytkownika)
    \item[BCP]     Business Continuity Plan (plan ciągłości działania)
    \item[CISO]    Chief Information Security Officer (dyrektor ds. bezpieczeństwa informacji)
    \item[CSIRT]   Computer Security Incident Response Team (zespół reagowania na incydenty bezpieczeństwa komputerowego)
    \item[DORA]    Digital Operational Resilience Act (rozporządzenie w sprawie operacyjnej odporności cyfrowej sektora finansowego)
    \item[EDR]     Endpoint Detection and Response (system wykrywania i reagowania na zagrożenia na stacjach końcowych)
    \item[IOD]     Inspektor Ochrony Danych
    \item[IRP]     Incident Response Plan (plan reagowania na incydenty)
    \item[ISO/IEC] International Organization for Standardization / International Electrotechnical Commission
    \item[KNF]     Komisja Nadzoru Finansowego
    \item[KPI]     Key Performance Indicator (kluczowy wskaźnik efektywności)
    \item[KYC]     Know Your Customer (procedura należytej staranności wobec klienta)
    \item[MFA]     Multi-Factor Authentication (uwierzytelnianie wieloskładnikowe)
    \item[MTTD]    Mean Time to Detect (średni czas wykrycia incydentu)
    \item[MTTR]    Mean Time to Respond / Resolve (średni czas reakcji / rozwiązania incydentu)
    \item[NIST]    National Institute of Standards and Technology (Krajowy Instytut Standaryzacji i Technologii USA)
    \item[PCI DSS] Payment Card Industry Data Security Standard (standard bezpieczeństwa danych w branży kart płatniczych)
    \item[PSD2]    Payment Services Directive 2 (dyrektywa w sprawie usług płatniczych w ramach rynku wewnętrznego)
    \item[RACI]    Responsible, Accountable, Consulted, Informed (macierz przypisania odpowiedzialności)
    \item[RCA]     Root Cause Analysis (analiza przyczyn źródłowych)
    \item[RODO]    Rozporządzenie o Ochronie Danych Osobowych (Rozporządzenie Parlamentu Europejskiego i Rady (UE) 2016/679)
    \item[SIEM]    Security Information and Event Management (system zarządzania informacjami i zdarzeniami bezpieczeństwa)
    \item[SLA]     Service Level Agreement (umowa o gwarantowanym poziomie świadczenia usług)
    \item[SOC]     Security Operations Center (centrum operacji bezpieczeństwa)
    \item[SZBI]    System Zarządzania Bezpieczeństwem Informacji
    \item[UODO]    Urząd Ochrony Danych Osobowych
\end{abbreviations}
